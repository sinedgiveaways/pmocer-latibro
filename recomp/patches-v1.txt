# rec-null-guard: the record walk in f_7001759b20 (main+0x1759b20) reads a per-record sub-table
# pointer that is null while the record is still being initialized (sub-record race, seen in the
# 22/11 crash log: the load ldr w9, [x25, #0x5c] faults at 0x104 from a null base, and in the
# 23/11 build-278 log a wild (2 - w8) * 0x38 index walked out of the patch page). The guard clamps
# the address instead: when the base is null, the load reads [x27] itself (0), and the function's
# own cbz then skips the empty element, the same skip the SDK code applies to the -1 state
# elsewhere (main+0x175a22c). When the base is valid, the computed address passes untouched. On
# the console the pointer is never null, so the switch stays on by default.
#
#   main+0x175a01c: 69 03 40 f9   ldr  x9, [x27]        ; sub-table pointer
#   main+0x175a030: 29 5f 40 b9   ldr  w9, [x25, #0x5c] ; faulted when [x27] = 0
#   main+0x175a034: 69 eb ff 34   cbz  w9, #0x1759da0   ; skips a zero word
#
# Replacement at main+0x175a030 (the faulting load; x9 still holds the base there):
#   neg  x13, x9          ; -base
#   orr  x13, x13, x9     ; base | -base: 0 only when the base is null
#   lsr  x13, x13, #63    ; x13 = 1 when the base is valid, else 0
#   sub  x13, x13, #1     ; the run-time mask: 0 (valid base) or ~0 (null base)
#   sub  x14, x27, #0x5c  ; SAFE: reading [x14 + 0x5c] reads [x27] itself, mapped by definition
#   bic  x16, x25, x13    ; the computed address when the base is valid
#   and  x17, x14, x13    ; SAFE when the base is null
#   orr  x25, x16, x17    ; the address the load below actually uses
#   ldr  w9, [x25, #0x5c] ; the original load, now safe
# x13, x14 and x16/x17 are scratch on this stretch (nothing between the load and the cbz reads them).
